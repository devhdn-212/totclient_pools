-- 1. Fungsi untuk Level dan Path Awal
CREATE OR REPLACE FUNCTION db_bbca.trg_item_category_logic()
RETURNS TRIGGER AS $$
DECLARE
    v_parent_path TEXT;
    v_parent_level INT;
BEGIN
    IF NEW.parent_id IS NULL THEN
        NEW.level := 1;
        NEW.path := ''; -- JANGAN diisi Name
    ELSE
        -- Ambil path parent (isinya angka, misal '1/5')
        SELECT path, level INTO v_parent_path, v_parent_level 
        FROM db_bbca.tbl_item_category WHERE id = NEW.parent_id;
        
        NEW.level := v_parent_level + 1;
        NEW.path := v_parent_path; 
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 2. Fungsi untuk Menambahkan ID ke Path (Suffix)
CREATE OR REPLACE FUNCTION db_bbca.trg_item_category_path_suffix()
RETURNS TRIGGER AS $$
BEGIN
    -- Update path: jika kosong jadi 'ID', jika ada isinya jadi 'parent_path/ID'
    EXECUTE format(
        'UPDATE %I.%I SET path = CASE WHEN path = '''' THEN id::TEXT ELSE path || ''/'' || id::TEXT END WHERE id = $1', 
        TG_TABLE_SCHEMA, 
        TG_TABLE_NAME
    ) USING NEW.id;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- Pasang trigger sebelum INSERT dan UPDATE
CREATE TRIGGER trg_1_before_upsert 
BEFORE INSERT OR UPDATE OF parent_id ON db_bbca.tbl_item_category 
FOR EACH ROW EXECUTE FUNCTION db_bbca.trg_item_category_logic();

CREATE TRIGGER trg_2_after_insert 
AFTER INSERT ON db_bbca.tbl_item_category 
FOR EACH ROW EXECUTE FUNCTION db_bbca.trg_item_category_path_suffix();

CREATE OR REPLACE FUNCTION db_bbca.trg_item_category_status_sync()
RETURNS TRIGGER AS $$
BEGIN
    -- Jika status berubah dari Y (Aktif) ke N (Tidak Aktif)
    IF OLD.status = 'Y' AND NEW.status = 'N' THEN
        -- Nonaktifkan semua sub-kategori di bawahnya berdasarkan PATH
        -- Contoh: Makanan (path: 12) jadi N, maka semua path '12/%' ikut jadi N
        EXECUTE format(
            'UPDATE %I.%I SET status = ''N'' WHERE path LIKE $1 || ''/%%'' OR path = $1', 
            TG_TABLE_SCHEMA, TG_TABLE_NAME
        ) USING OLD.path;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Pasang Trigger
CREATE TRIGGER trg_3_status_sync
AFTER UPDATE OF status ON db_bbca.tbl_item_category
FOR EACH ROW EXECUTE FUNCTION db_bbca.trg_item_category_status_sync();

-- 1. Fungsi Utama untuk Hitung Path dan Level
CREATE OR REPLACE FUNCTION db_bbca.fn_recursive_category_update()
RETURNS TRIGGER AS $$
DECLARE
    parent_path TEXT;
    parent_level INT;
BEGIN
    -- Jika tidak punya parent (Root)
    IF NEW.parent_id IS NULL THEN
        NEW.path := NEW.id::text;
        NEW.level := 1;
    ELSE
        -- Ambil path dan level dari parent-nya
        SELECT path, level INTO parent_path, parent_level 
        FROM db_bbca.tbl_item_category 
        WHERE id = NEW.parent_id;
        
        NEW.path := parent_path || '/' || NEW.id;
        NEW.level := parent_level + 1;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 2. Trigger BEFORE untuk menjamin data Path & Level selalu benar sebelum disimpan
-- Hapus dulu jika sudah ada nama yang sama
DROP TRIGGER IF EXISTS trg_item_category_path_logic ON db_bbca.tbl_item_category;

CREATE TRIGGER trg_item_category_path_logic
BEFORE INSERT OR UPDATE OF parent_id ON db_bbca.tbl_item_category
FOR EACH ROW
EXECUTE FUNCTION db_bbca.fn_recursive_category_update();
